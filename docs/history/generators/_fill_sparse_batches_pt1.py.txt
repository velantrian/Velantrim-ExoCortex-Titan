def count_rows(text):
    return sum(1 for l in text.splitlines() if l.startswith('|') and not l.startswith('|---') and 'ID |' not in l)

def write_patch(filename, new_facts):
    path = RU / filename
    if not path.exists():
        print(f"MISSING {filename}")
        return 0
    text = path.read_text(encoding='utf-8')
    lines = text.splitlines()
    # Keep header + separator
    out_lines = []
    header_area = True
    added = 0
    for i, line in enumerate(lines):
        if line.startswith('|') and 'ID |' in line:
            header_area = False
            out_lines.append(line)
            continue
        if not header_area:
            if line.startswith('|---'):
                out_lines.append(line)
                # insert new rows right after separator
                ns_match = re.search(r'\*\*Namespace:\*\* `(.+?)\.\*`', text)
                ns = ns_match.group(1) if ns_match else 'unknown'
                for suffix, title, typ, essence, practical in new_facts:
                    out_lines.append(f"| {ns}.{suffix} | {title} | {typ} | {essence} | {practical} |")
                    added += 1
                header_area = True  # done inserting
            else:
                out_lines.append(line)
        else:
            out_lines.append(line)
    new_text = '\n'.join(out_lines)
    # Update KnowledgeUnits count
    new_text = re.sub(r'(\*\*KnowledgeUnits:\*\*\s*)\d+', rf'\g<1>{count_rows(new_text)}', new_text, count=1)
    path.write_text(new_text, encoding='utf-8')
    print(f"  {filename}: +{added} фактов → всего {count_rows(new_text)}")
    return added

if __name__ == '__main__':
    total = 0
    for fname, facts in FILL_FACTS.items():
        total += write_patch(fname, facts)
    print(f"\n✅ Итого: +{total} фактов")
